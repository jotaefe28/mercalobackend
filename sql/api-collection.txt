# MercaloPOS API Collection - Postman
# Sistema POS Multitenant con Sistema de Puntos (1 punto = $1)
# ================================================================

## Base URL
http://localhost:3000/api

## Variables de Entorno Postman
- {{base_url}} = http://localhost:3000/api
- {{access_token}} = [Se obtendrá del login]
- {{company_id}} = [Se obtendrá del login]
- {{user_id}} = [Se obtendrá del login]

# ================================================================
# 1. AUTENTICACIÓN (/api/auth)
# ================================================================

### 1.1 Registro de Nueva Empresa
POST {{base_url}}/auth/register
Content-Type: application/json

{
  "company": {
    "name": "Mi Nueva Tienda POS",
    "tax_id": "987654321-1",
    "email": "admin@nuevatienda.com",
    "phone": "+57 300 123 4567",
    "address": "Calle 123 #45-67, Ciudad"
  },
  "user": {
    "name": "Administrador Principal",
    "email": "admin@nuevatienda.com",
    "password": "Admin123456!",
    "confirmPassword": "Admin123456!"
  }
}

### respuesta ok 
{
    "success": true,
    "message": "Recurso creado exitosamente",
    "data": {
        "company": {
            "id": "9715672a-1a30-424d-8906-0c99b9014017",
            "name": "Mi Nueva Tienda POS",
            "plan": "BASIC"
        },
        "user": {
            "id": "ceda14a9-66a0-488b-b338-9a69e5eb8d81",
            "email": "admin@nuevatienda.com",
            "name": "Administrador Principal",
            "role": "ADMIN"
        }
    }
}


### 1.2 Login
POST {{base_url}}/auth/login
Content-Type: application/json

{
  "email": "admin@mitienda.com",
  "password": "Admin123456!"
}

### 1.3 Refresh Token
POST {{base_url}}/auth/refresh
Content-Type: application/json

{
  "refreshToken": "{{refresh_token}}"
}

### 1.4 Logout
POST {{base_url}}/auth/logout
Authorization: Bearer {{access_token}}

### 1.5 Cambiar Contraseña
PUT {{base_url}}/auth/change-password
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "currentPassword": "Admin123456!",
  "newPassword": "NewPassword123!",
  "confirmPassword": "NewPassword123!"
}

### 1.6 Recuperar Contraseña
POST {{base_url}}/auth/forgot-password
Content-Type: application/json

{
  "email": "admin@mitienda.com"
}

POST {{base_url}}/users
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "name": "Juan Pérez",
  "email": "juan@mitienda.com",
  "password": "Juan123456!",
  "role": "CASHIER",
  "phone": "+57 301 234 5678",
  "is_active": true
}

### 2.2 Listar Usuarios
GET {{base_url}}/users?page=1&limit=10&role=CASHIER&is_active=true
Authorization: Bearer {{access_token}}

### 2.3 Obtener Usuario por ID
GET {{base_url}}/users/{{user_id}}
Authorization: Bearer {{access_token}}

### 2.4 Actualizar Usuario
PUT {{base_url}}/users/{{user_id}}
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "name": "Juan Pérez Actualizado",
  "phone": "+57 301 234 5679",
  "role": "MANAGER"
}

### 2.5 Eliminar Usuario
DELETE {{base_url}}/users/{{user_id}}
Authorization: Bearer {{access_token}}

### 2.6 Activar/Desactivar Usuario
PATCH {{base_url}}/users/{{user_id}}/toggle-status
Authorization: Bearer {{access_token}}

# 6. SISTEMA DE PUNTOS (/api/points) - 1 punto = $1
# ================================================================

### 6.1 Consultar Balance de Puntos
GET {{base_url}}/points/balance/{{client_id}}
Authorization: Bearer {{access_token}}

### 6.2 Agregar Puntos Manualmente
POST {{base_url}}/points/add
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "client_id": "{{client_id}}",
  "points": 100,
  "description": "Puntos de bonificación por cumpleaños",
  "expires_at": "2025-10-21T23:59:59Z"
}

### 6.3 Redimir Puntos
POST {{base_url}}/points/redeem
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "client_id": "{{client_id}}",
  "points": 50,
  "description": "Redención de puntos en compra"
}

### 6.4 Transferir Puntos
POST {{base_url}}/points/transfer
Authorization: Bearer {{access_token}}
Content-Type: application/json

{
  "from_client_id": "{{client_id_1}}",
  "to_client_id": "{{client_id_2}}",
  "points": 25,
  "description": "Transferencia de puntos entre clientes"
}

### 6.5 Historial de Puntos del Cliente
GET {{base_url}}/points/history/{{client_id}}?startDate=2024-01-01&endDate=2024-12-31&type=REDEEMED&limit=50
Authorization: Bearer {{access_token}}

### 6.6 Historial General de Puntos
GET {{base_url}}/points/history?limit=100&offset=0
Authorization: Bearer {{access_token}}

### 6.7 Resumen de Puntos
GET {{base_url}}/points/summary
Authorization: Bearer {{access_token}}

### 6.8 Puntos Por Expirar
GET {{base_url}}/points/expiring
Authorization: Bearer {{access_token}}

### 6.9 Configuración de Puntos
GET {{base_url}}/points/config
Authorization: Bearer {{access_token}}
